```tsx
import siteMetadata from '@/data/siteMetadata'
import headerNavLinks from '@/data/headerNavLinks'
import Link from './Link'
import MobileNav from './MobileNav'

const Header = () => {
  // ==============================================================================
  // 【液态玻璃（Liquid Glass）核心样式配置】
  // ==============================================================================
  let headerClass = [
    // ----------------------------------------------------------------------------
    // 1. 基础盒模型与尺寸（两端贴边与紧凑高度）
    // ----------------------------------------------------------------------------
    'flex w-full items-center justify-between',
    // px-6 sm:px-12：左右两侧留白，避免文字生硬贴死屏幕物理边缘
    'px-6 sm:px-12',
    // py-3：大幅压低顶部导航高度，保持干练紧凑
    'py-3',
    // 状态切换（如吸顶激活）时的过渡平滑度
    'transition-all duration-300 ease-out',

    // ----------------------------------------------------------------------------
    // 2. 极致透光底色（摒弃传统白雾磨砂，改用水膜垂直渐变）
    // ----------------------------------------------------------------------------
    // 浅色模式：顶部仅保留 8% 微弱反光，向下快速衰减至 1% 乃至完全透明
    'bg-gradient-to-b from-white/[0.08] via-white/[0.01] to-transparent',
    // 深色模式：顶部 6% 白光反衬，向下归零
    'dark:from-white/[0.06] dark:via-white/[0.005] dark:to-transparent',

    // ----------------------------------------------------------------------------
    // 3. 液体折射光学滤镜（Backdrop Filter）
    // ----------------------------------------------------------------------------
    // backdrop-blur-[2.5px]：极轻微模糊，保留水体微折射的同时彻底杜绝磨砂塑料感
    'backdrop-blur-[2.5px]',
    // backdrop-saturate-[210%]：水滴透镜效应，将穿透的底层文字与背景饱和度提升，色彩更晶莹
    'backdrop-saturate-[210%]',
    // backdrop-contrast-[103%]：微调穿透对比度，保证高透状态下文字清晰度
    'backdrop-contrast-[103%]',

    // ----------------------------------------------------------------------------
    // 4. 水滴表面张力高光线（顶部 Specular Highlight）
    // ----------------------------------------------------------------------------
    // inset 0 1px 0 0：在顶部内侧绘制极细的高光水线，模拟自然环境光照射在水膜顶缘的反光
    'shadow-[inset_0_1px_0_0_rgba(255,255,255,0.75),0_8px_32px_rgba(0,0,0,0.015)]',
    'dark:shadow-[inset_0_1px_0_0_rgba(255,255,255,0.12)]',

    // ----------------------------------------------------------------------------
    // 5. 弱化下边框分界线（Subtle Bottom Border）
    // ----------------------------------------------------------------------------
    // 启用 1px 下边框
    'border-b',
    // 【关键弱化参数】：
    // 将不透明度压低至 1.5%（border-black/[0.015]）与深色 3%（border-white/[0.03]）
    // 远看近乎隐形，仅在滚动正文内容穿过时隐约勾勒出水膜下边缘的微弱分界
    // 若希望彻底移除，可直接将此行替换为 'border-b-0' 或删除
    'border-black/[0.001] dark:border-white/[0.001]',
  ].join(' ')

  // ------------------------------------------------------------------------------
  // 6. 吸顶逻辑注入
  // ------------------------------------------------------------------------------
  if (siteMetadata.stickyNav) {
    // sticky top-0 z-50：确保页面滚动时固定在最顶层，不会被后续正文遮挡
    headerClass += ' sticky top-0 z-50'
  }

  return (
    <header className={headerClass}>
      {/* ==========================================================================
          【左侧品牌标识区】：紧贴视口左侧，无多余装饰
         ========================================================================== */}
      <Link href="/" aria-label={siteMetadata.headerTitle}>
        <span
          className="text-base font-semibold tracking-tight text-gray-900 transition-opacity hover:opacity-60 dark:text-gray-100 sm:text-lg"
        >
          {siteMetadata.headerTitle}
        </span>
      </Link>

      {/* ==========================================================================
          【右侧导航与操作区】：紧贴视口右侧，包含桌面链接与移动端折叠菜单
         ========================================================================== */}
      <div className="flex items-center space-x-6">
        {/* 桌面端导航列表（根据 data/headerNavLinks 渲染 Blogs、About） */}
        <div className="no-scrollbar hidden items-center space-x-6 sm:flex">
          {headerNavLinks
            .filter((link) => link.href !== '/')
            .map((link) => (
              <Link
                key={link.title}
                href={link.href}
                className="text-sm font-medium text-gray-700 transition-opacity hover:opacity-50 dark:text-gray-200 dark:hover:opacity-60 sm:text-base"
              >
                {link.title}
              </Link>
            ))}
        </div>

        {/* 移动端汉堡菜单（屏幕宽度较小时自动显示） */}
        <MobileNav />
      </div>
    </header>
  )
}

export default Header
```
